Google Form Auto-Import

Connect a Google Form to TalentFlow so every new submission is automatically added to your pipeline as an "On Hold" candidate.

1

Create a Google Form

Add fields: Name, Email, Phone, Position, Department, Experience, Skills, Notes.

2

Open Apps Script

In your Form, go to Extensions → Apps Script. Paste the script below.

3

Set trigger

Add a form submit trigger pointing to onFormSubmit and deploy.

Your Credentials (auto-injected into script)

Apps Script (onFormSubmit) 
function onFormSubmit(e) {
  var USER_EMAIL          = "YOUR_EMAIL";
  var USER_PASSWORD       = "YOUR_PASSWORD";
  var API_KEY             = "AIzaSyB3Z4-HU5pADplI-5ONrnV-4sm-eYMXRLo";
  var PROJECT_ID          = "ats-system-d83ef";

  // Step 1: Sign in to get ID token
  var signInResp = UrlFetchApp.fetch(
    "https://identitytoolkit.googleapis.com/v1/accounts:signInWithPassword?key=" + API_KEY,
    {
      method: "post",
      contentType: "application/json",
      payload: JSON.stringify({ email: USER_EMAIL, password: USER_PASSWORD, returnSecureToken: true }),
      muteHttpExceptions: true
    }
  );
  var signInData = JSON.parse(signInResp.getContentText());
  var idToken    = signInData.idToken;
  var uid        = signInData.localId;
  if (!idToken) { Logger.log("Auth failed: " + signInResp.getContentText()); return; }

  // Step 2: Map form fields
  var r = e.namedValues;
  var name     = (r["Full Name"]    || r["Name"]       || [""])[0].trim();
  var email    = (r["Email"]        || r["Email Address"] || [""])[0].trim();
  var phone    = (r["Phone"]        || r["Phone Number"]  || [""])[0].trim();
  var position = (r["Position"]     || r["Role"]          || [""])[0].trim();
  var dept     = (r["Department"]   || [""])[0].trim();
  var exp      = (r["Experience"]   || r["Years of Experience"] || ["0"])[0].trim();
  var skills   = (r["Skills"]       || [""])[0].trim();
  var notes    = (r["Notes"]        || r["Additional Info"] || [""])[0].trim();

  var docBody = {
    fields: {
      name:      { stringValue: name },
      nameLower: { stringValue: name.toLowerCase() },
      email:     { stringValue: email },
      phone:     { stringValue: phone },
      position:  { stringValue: position },
      dept:      { stringValue: dept },
      experience:{ stringValue: exp },
      skills:    { stringValue: skills },
      notes:     { stringValue: notes },
      status:    { stringValue: "On Hold" },
      source:    { stringValue: "Google Form" },
      createdAt: { timestampValue: new Date().toISOString() }
    }
  };

  // Step 3: Write to Firestore using Bearer token
  var url = "https://firestore.googleapis.com/v1/projects/" + PROJECT_ID
          + "/databases/(default)/documents/users/" + uid + "/candidates";

  var resp = UrlFetchApp.fetch(url, {
    method: "post",
    contentType: "application/json",
    headers: { "Authorization": "Bearer " + idToken },
    payload: JSON.stringify(docBody),
    muteHttpExceptions: true
  });
  Logger.log("Firestore response: " + resp.getContentText());
}
